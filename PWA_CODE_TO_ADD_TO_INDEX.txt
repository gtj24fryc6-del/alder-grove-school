PASTE THESE LINES INSIDE THE <head> ... </head> SECTION OF index.html

<link rel="manifest" href="/alder-grove-school/manifest.webmanifest">
<meta name="theme-color" content="#174D3A">
<link rel="apple-touch-icon" href="/alder-grove-school/apple-touch-icon.png">
<meta name="apple-mobile-web-app-capable" content="yes">
<meta name="apple-mobile-web-app-title" content="Alder Grove">

PASTE THIS SCRIPT JUST BEFORE THE CLOSING </body> TAG IN index.html

<script>
  if ("serviceWorker" in navigator) {
    window.addEventListener("load", function () {
      navigator.serviceWorker.register("/alder-grove-school/service-worker.js")
        .catch(function (error) {
          console.log("App offline setup could not be registered:", error);
        });
    });
  }
</script>
